-- 사원 정보 데이터를 담은 테이블
CREATE TABLE employees(
	employee_id		number(6,0)		PRIMARY KEY, 	    -- 사원번호
	emp_name		varchar2(80)    NOT NULL, 			-- 사원명 
	email			varchar2(100),            			-- 사원 이메일 , 이메일주소가 너무 길 경우 효율성이 떨어짐 
	phone_number	varchar2(20),			  			-- 전화번호 
	hire_date		date			NOT NULL, 			-- 입사일자
	salary			number(8,2),              			-- 급여
	manager_id		number(6,0),              			-- 상위 관리자 번호
	commission_pct	number(2,2),			  			-- 커미션율
	retire_date     date,                     			-- 퇴사일자
	department_id   number(6,0),              			-- 소속된 부서번호
	job_id			varchar2(10),             			-- 직무id
	create_date		DATE            DEFAULT SYSDATE,    -- 데이터 생성날짜
	-- 데이터 수정날짜
	update_date     DATE            DEFAULT SYSDATE     
);


-- employees 더블클릭해서 나오는 화면에서 코멘트를 보고싶을때

COMMENT ON COLUMN EMPLOYEES.email IS '이메일';
COMMENT ON COLUMN EMPLOYEES.phone_number IS '전화번호';
COMMENT ON COLUMN EMPLOYEES.hire_date IS '입사일자';
COMMENT ON COLUMN EMPLOYEES.salary IS '급여';
COMMENT ON COLUMN EMPLOYEES.manager_id IS '상위 관리자 번호';
COMMENT ON COLUMN EMPLOYEES.commission_pct IS '커미션율';
COMMENT ON COLUMN EMPLOYEES.retire_date IS '퇴사일자';
COMMENT ON COLUMN EMPLOYEES.department_id IS '소속된 부서번호';
COMMENT ON COLUMN EMPLOYEES.job_id IS '직무id';
COMMENT ON COLUMN EMPLOYEES.create_date IS '데이터 생성날짜';
COMMENT ON COLUMN EMPLOYEES.update_date IS '데이터 수정날짜';

-- SALARY컬럼의 값이 5000보다 큰 행 중에서 사원번호와 사원이름 값만 가져오기 (58개의 행이 나옴)

  FROM employees
 WHERE salary > 5000; 

-- salary컬럼의 값이 5000보다 큰 행 중에서 사원번호와 사원이름 값만 가지고 오면서 employee_id의 값이 오름차순으로 조회
SELECT employee_id, emp_name, job_id
  FROM employees
 WHERE salary > 5000
 ORDER BY employee_id ASC   -- asc는 오름차순, desc는 내림차순
;  -- order by는 첫번째 값이 같을때만 두번쨰 값을 보는데 지금은 첫번째가 기본키로 되어있기 때문에 중복될 수가 없어서 결과는 의미없음

-- AND조선 사용(조건식이 모두 맞아야 함)
SELECT employee_id, emp_name
  FROM employees
 WHERE salary > 5000
  AND job_id = 'IT_PROG'    --IT_PROG를 소문자로 하면 안됨  두가지 조건을 만족한다는 의미 
 ORDER BY employee_id ASC
;

-- OR조건 사용(조건식이 1개만 맞아도 조회가 가능(급여가 5000이거나 직무가 IT_PROG인 데이터))
SELECT employee_id, emp_name, salary, job_id
  FROM employees
 WHERE salary > 5000
    OR job_id = 'IT_PROG'
 ORDER BY employee_id asc
;

-- 닉네임을 선언해서 as 닉네임명 으로 사용하는 경우 
SELECT employee_id AS b, emp_name AS c, salary AS d, job_id
  FROM employees   -- 테이블은 닉네임으로 사용 못함 이건 두개 이상일때 사용가능함 
 WHERE salary > 5000
    OR job_id = 'IT_PROG'
 ORDER BY employee_id asc
;

SELECT employee_id, salary
  FROM employees
 WHERE salary = ANY (2000, 3000, 4000)
 ORDER BY employee_id ASC
;

-- any랑 같은 결과가 나옴 ('or' 사용)
SELECT employee_id, salary
  FROM employees
 WHERE salary = 2000
    OR salary = 3000
    OR salary = 4000
 ORDER BY employee_id ASC
;

SELECT employee_id, salary
  FROM employees
 WHERE salary = all(2000, 3000, 4000)
 ORDER BY employee_id ASC
;

SELECT employee_id, salary

   and salary = 3000
   and salary = 4000

-- department부서 테이블 생성
CREATE TABLE departments(
	department_id    		number(6,0)      PRIMARY KEY,    -- 부서번호
	department_name  		varchar2(80)	 NOT NULL,       -- 부서이름
	manager_id   			number(6,0),                     -- 부서장의 employee_id
	parent_id               number(6,0),                     -- 상위부서 번호
	create_date      		DATE          DEFAULT sysdate,   -- 데이터 생성날짜
	-- 데이터 수정날짜
	update_date      		DATE          DEFAULT sysdate   
);

COMMENT ON COLUMN departments.department_id IS '부서번호';
COMMENT ON COLUMN departments.department_name IS '부서이름';
COMMENT ON COLUMN departments.manager_id IS '부서장의 employee_id';
COMMENT ON COLUMN departments.parent_id IS '상위부서 번호';
COMMENT ON COLUMN departments.create_date IS '데이터 생성날짜';
COMMENT ON COLUMN departments.update_date IS '데이터 수정날짜';

-- exists조건식 사용(임시 테이블(exist다음의 ()를 의미) 정보의 데이터가 있는지 체크해주는 조건식)
-- 급여가 3000이상을 받은 직원이 있는 부서의 이름 조회  , 부서는 다른 테이블에 있기 때문에 부서 id로 연결하는거임
SELECT department_id, department_name  
  FROM departments a     -- 부서아이디만 있기 떄문에 부서명을 가져오기 위해서 
 -- exists 다음에 ()는 서브쿼리라고 함
 WHERE exists(SELECT * 
 				FROM employees b 
 			   WHERE a.DEPARTMENT_ID = b.department_id 
 			     AND b.salary > 3000
 	   )
ORDER BY a.department_name
;
-- 문법상 오류는 아닌데 1000번이 없는 경우 
SELECT department_id, department_name
  FROM departments a
 WHERE exists(SELECT department_id  FROM departments WHERE department_id = 1000)
;

-- LIKE 조건식(직원 이름이 'Al'로 시작하는 모든 직원 데이터 조회, '%'는 모든 글자들을 의미함)
SELECT emp_name
  FROM employees
 WHERE emp_name LIKE 'Al%'
 ORDER BY emp_name
;   

-- LIKE 조건식(직원 이름이 'z'로 끝나는 모든 직원 데이터 조회, '%'는 모든 글자들을 의미함)
SELECT emp_name
  FROM employees
 WHERE emp_name LIKE '%z'
 ORDER BY emp_name
;  

-- LIKE 조건식(직원 이름이 처음에 'D'로 시작하고 중간에 't'가 있는 모든 직원 데이터 조회, '%'는 모든 글자들을 의미함)
SELECT emp_name
  FROM employees
 WHERE emp_name LIKE 'D%t%'
 ORDER BY emp_name
;
