--      col2의 값은 10, col3의 값은 현재시간으로 변경     --> 어차피 col2 = 30 AND col1 = 'GHI';를 만족하는 값이 없기 때문에 기존의 값을 유지
UPDATE ex3_1 SET col2 = 10, col3 = SYSDATE WHERE col2 = 30 AND col1 = 'GHI';

-- 문제: col1이 '10'(문자열)이면서(and) col2가 10인 행의 col1 값을 'TEN'으로 변경하시오. (기존 값 유지)
UPDATE ex3_1 SET col1 = 'TEN' WHERE col1 = '10' AND col2 = 10;

-- MERGE문
--   MERGE INTO [스키마명].테이블명 USING (데이터의 원천(SELECT)) ON insert or update될 조건
--   	WHEN MATCHED THEN SET 컬럼1 = 값1, 컬럼2 = 값2, ... WHERE update조건
--          DELETE WHERE update및delete 조건
--      WHEN NOT MATCHED THEN INSERT (컬럼1, 컬럼2, ...) VALUES(값1, 값2, ... WHERE insert조건

	prod_id				number(6,0)		NOT NULL,		-- 상품ID(번호)

-- 오라클에서 지원하는 함수 모음
-- 1. 숫자 함수
-- (1) abs(n)함수: 절대값을 반환하는 함수 
SELECT abs(-1) abs1 FROM dual; -- abs1이라는 컬럼으로 절대값 반환
SELECT abs(-1) AS abs1 FROM dual; 
SELECT abs(10), abs(-10), abs(-10.123) FROM dual;

-- (2) ceil(n)함수: 올림함수
SELECT ceil(10.123), ceil(10.541), ceil(11.001) FROM dual;

-- (3) floor(n)함수: 내림함수
SELECT floor(10.123), floor(10.541), floor(11.001) FROM dual;

-- (4) round(n, i)함수: 반올림함수(n은 실제값, i는 소수점 기준 i+1번째에 반올림하기 위한 값 (i는 생략가능))
SELECT round(10.154), round(10.541), round(11.001) FROM dual;  -- i가 없는 소수점 첫번째자리 기준  
SELECT ROUND(10.154, 1), ROUND(10.154, 2), ROUND(10.154, 3) FROM DUAL; 
SELECT ROUND(0, 3), ROUND(115.155, -1), ROUND(115.155, -2) FROM DUAL;  

-- (5) trunc(n1, n2)함수: 반올림하지 않고 n1을 소수점기준 n2자리에서 무조건 잘라낸 결과를 반환
SELECT trunc(115.155), trunc(115.155,1), trunc(115.155,2), trunc(115.155,-2) FROM dual;

-- (6) power(n2, n1)함수: n2를 n1제곱한 결과를 반환
SELECT power(3,2), power(3,3), power(3,3.0001) FROM dual;

-- (7) sqrt(n)함수: n의 제곱근을 반환
select sqrt(2), sqrt(5) FROM dual;

-- (8) mod(n2, n1)함수: n2를 n1으로 나눈 나머지값을 반환
select mod(19,4), mod(19.123, 4.2) FROM dual; --19 - 4 * FLOOR(19 / 4) = 19 - 4*4 = 3

-- (9) remainder(n2, n1)함수: 나머지를 구하는 내부적 연산 방법이 mod함수랑 다름
SELECT remainder(19,4), remainder(19.123, 4.2) FROM dual;  --19 - 4 * ROUND(19 / 4) = 19 - 4*5 = -1

-- (10) exp(n)함수: 지수함수로 e의 n제곱 값을 반환
-- (11) ln(n)함수: 자연 로그 함수로 밑수가 e인 로그 함수
-- (12) log(n2, n1)함수: n2를 킽수로 하는 n1의 로그값을 반환
SELECT exp(2), ln(2.713), log(10,100) FROM dual;

-- 2. 문자 함수 
-- (1) initcap(char)함수: 매개변수로 들어오는 char의 첫 문자는 대문자로, 나머지는 소문자로 반환
SELECT initcap('never say goodbye'), initcap('never6say*good가bye') FROM dual;

-- (2) lower(char)함수: 매개변수로 들어오는 문자를 모두 소문자로 변환
-- (3) upper(char)함수: 매개변수로 들어오는 문자를 대문자로 변환
SELECT lower('NEVER SAY GOODBYE'), UPPER('never say goodbye') FROM dual;

-- (4) concat(char1, char2)함수: '||'연산자처럼 매개변수로 들어오는 두 문자를 붙여서 반환
SELECT concat('I Have', ' A dream'), 'I Have' || ' A dream' FROM dual;

-- (5) substr(char, pos, len)함수: 잘라올 대사 문자열인 char의 pos번째 문자부터 len길이만큼 잘라낸 결과를 반한하는 함수
SELECT substr('ABCDEFG', 1, 4), substr('ABCDEFG', -1, 4) FROM dual;

-- (6) substrb(char, pos, len)함수: 문자 개수가 아닌 문자열의 바이트수만큼 잘라낸 결과를 반환
select substrb('ABCDEFG', 1, 4), substrb('가나다라마바사',1,4) FROM dual; --한글은 UTF-8 기준으로 1자 = 3바이트

-- (7) ltrim(char,set)함수: 매개변수로 들어온 char문자열에서 set으로 지정된 문자열을 왼쪽 끝에서 제거한 후 나머지 문자열을 반환
-- (8) rtrim(char,set)함수: ltlim 함수와 반대로 오른쪽 끝에서 제거한 뒤 나머지 문자열을 반환하는 함수
SELECT ltrim('ABCDEFGABC', 'ABC'),
	   ltrim('가나다라', '가'),
	   rtrim('ABCDEFGABC', 'ABC'),
	   rtrim('가나다라', '라') FROM dual;
